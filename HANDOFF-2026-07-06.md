```cpp
    StaticJsonDocument<256> doc;
    doc["current_transport"] = (int)current;
    doc["wifi_rssi"] = metrics.rssi;
    doc["wifi_connected"] = network.getCurrentTransport() == TRANSPORT_WIFI;
    doc["packets_sent"] = metrics.packets_sent;
    doc["failures"] = metrics.failures;

    String json;
    serializeJson(doc, json);
    webServer.send(200, "application/json", json);
});
```

```cpp
#ifndef OTA_MANAGER_H

#include <esp_ota_ops.h>
#include <mbedtls/pk.h>
#include <mbedtls/sha256.h>
#include "Packet.h"

namespace lorapnp {

enum OTAState {
    OTA_IDLE,
    OTA_DOWNLOADING,
    OTA_VERIFYING,
    OTA_APPLYING,
    OTA_COMPLETE,
    OTA_FAILED
};

struct OTASession {
    uint32_t session_id;
    uint16_t total_chunks;
    uint16_t chunks_received;
    uint8_t* chunk_bitmap;      // Dynamic allocation
    size_t bitmap_size;
    uint8_t firmware_sha256[32];
    uint32_t last_update_ms;
    OTAState state;
};

class OTAManager {
public:
    OTAManager();
    ~OTAManager();

    bool begin();
    void update();

    // Signature verification (RSA-2048)
    bool verifyFirmwareSignature(const uint8_t* firmware, size_t len,
                                 const uint8_t* signature, size_t sig_len);

    // Health check for rollback protection
    bool performDiagnostics();

    // Handle OTA protocol messages
    void handleOTAAnnounce(const Packet* pkt);
    void handleOTAChunk(const Packet* pkt);
    void handleOTAComplete(const Packet* pkt);

    OTAState getState() { return session.state; }
    uint8_t getProgress();  // 0-100

private:
    OTASession session;
    esp_ota_handle_t ota_handle;
    const esp_partition_t* update_partition;

    // Embedded public key (set at compile time)
    const char* ota_public_key_pem;
};

} // namespace lorapnp

#endif
```

#### Step 5: Build Script for Delta Generation

**File**: `tools/generate_delta.py`

```python
#!/usr/bin/env python3
import sys
import hashlib
import struct

BLOCK_SIZE = 4096

def generate_delta(old_fw, new_fw, output):
    with open(old_fw, 'rb') as f:
        old_data = f.read()
    with open(new_fw, 'rb') as f:
        new_data = f.read()

    old_hash = hashlib.sha256(old_data).digest()
    new_hash = hashlib.sha256(new_data).digest()

    # Calculate changed blocks
    total_blocks = (len(new_data) + BLOCK_SIZE - 1) // BLOCK_SIZE
    changed_blocks = []

    for i in range(total_blocks):
        start = i * BLOCK_SIZE
        end = min(start + BLOCK_SIZE, len(new_data))

        old_block = old_data[start:end] if start < len(old_data) else b''
        new_block = new_data[start:end]

        if old_block != new_block:
            # Pad to BLOCK_SIZE
            padded = new_block + b'\xff' * (BLOCK_SIZE - len(new_block))
            changed_blocks.append((i, padded))

    # Write delta file
    with open(output, 'wb') as f:
        # Header: total_blocks(2) + changed_blocks(2) + old_hash(32) + new_hash(32)
        f.write(struct.pack('>HH', total_blocks, len(changed_blocks)))
        f.write(old_hash)
        f.write(new_hash)

        # Changed blocks: block_index(2) + data(4096) + crc16(2)
        for idx, data in changed_blocks:
            f.write(struct.pack('>H', idx))
            f.write(data)
            # CRC16 calculation (implement or use library)
            crc = 0  # TODO: Calculate actual CRC16
            f.write(struct.pack('>H', crc))

    print(f"Delta: {len(changed_blocks)}/{total_blocks} blocks changed ({100*len(changed_blocks)/total_blocks:.1f}%)")
    print(f"Delta size: {len(changed_blocks) * 4100 + 68} bytes")

if __name__ == '__main__':
    if len(sys.argv) != 4:
        print("Usage: generate_delta.py old.bin new.bin delta.bin")
        sys.exit(1)

    generate_delta(sys.argv[1], sys.argv[2], sys.argv[3])
```
